==== 4. Изменение: замена курса по выбору ====
$ docker exec practice01-neo4j cypher-shell -u neo4j -p mysecretpassword --format plain MATCH (s:Student {student_card_number: '23Б0101'})-[r:CHOSE]->(:LectureCourse {name: 'Бизнес-аналитика'}) DELETE r WITH s MATCH (c:LectureCourse {name: 'NoSQL и полиглотное хранение'}) MERGE (s)-[:CHOSE]->(c) SET s.updated = date() RETURN s.last_name AS student, [(s)-[:CHOSE]->(x) | x.name] AS electives
